import java.security.*;
import java.util.*;

public class SHA1 {

    public static void main(String[] args) throws Exception {

        Scanner sc = new Scanner(System.in);

        // Accept message
        System.out.print("Enter a message: ");
        String msg = sc.nextLine();

        // Create SHA-1 object
        MessageDigest md = MessageDigest.getInstance("SHA-1");

        //generate hash (160-bit)
        byte[] hash1 = md.digest(msg.getBytes());
        String originalHash = bytesToHex(hash1); //40 hex chars

        System.out.println("Original SHA-1: " + originalHash);

        // Modify message
        System.out.print("Enter modified message: ");
        String msg2 = sc.nextLine();

        byte[] hash2 = md.digest(msg2.getBytes());
        String modifiedHash = bytesToHex(hash2);

        System.out.println("Modified SHA-1: " + modifiedHash);

        if (originalHash.equals(modifiedHash)) {
            System.out.println("Messages are identical!");
        } else {
            System.out.println("Messages are different!");
        }
    }

    // Convert byte array to hexadecimal
    static String bytesToHex(byte[] bytes) {

        StringBuilder sb = new StringBuilder();

        for (byte b : bytes) {
            sb.append(String.format("%02X", b));
        }

        return sb.toString();
    }
}
